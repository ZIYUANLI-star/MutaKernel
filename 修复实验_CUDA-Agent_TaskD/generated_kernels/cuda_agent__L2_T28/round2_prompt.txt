class Model(nn.Module):
    """
    Model that performs a GEMM, GroupNorm, Swish, Multiply, and Swish operations.

        self.group_norm = nn.GroupNorm(num_groups, out_features)

        # (batch_size, out_features) -> (batch_size, out_features)
