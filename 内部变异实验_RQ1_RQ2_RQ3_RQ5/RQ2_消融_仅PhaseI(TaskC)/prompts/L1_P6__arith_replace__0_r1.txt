#include <torch/extension.h>
#include <cuda_runtime.h>
#include <cublas_v2.h>

torch::Tensor matmul_optimized_cuda(torch::Tensor A, torch::Tensor B) {
    cublasHandle_t handle;
    cublasCreate(&handle);
    
    int M = A.size(0);
    int K = A.size(1);
    int N = B.size(1);
    
    auto C = torch::zeros({M, N}, A.options());
    
    float alpha = 1.0f;
    float beta = 0.0f;
    
    // Use cuBLAS SGEMM for optimal performance
    cublasSgemm(handle,
                CUBLAS_OP_N, CUBLAS_OP_N,
                N, M, K,
                &alpha,
                B.data_ptr<float>(), N,
                A.data_ptr<float>(), K,
                &beta,
                C.data_ptr<float>(), N);
    
    cublasDestroy(handle);
    return C;
}
